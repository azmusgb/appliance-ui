// usage: node sweep.mjs <kit worktree> <out dir> ; needs playwright from the kit's node_modules and axe-core beside this file
import path from 'node:path'; import fs from 'node:fs'; import { pathToFileURL, fileURLToPath } from 'node:url'; import { createRequire } from 'node:module';
const [root, out] = process.argv.slice(2);
const here = path.dirname(fileURLToPath(import.meta.url));
const { chromium } = await import(pathToFileURL(path.join(root, 'node_modules/playwright/index.mjs')).href);
const axeSrc = fs.readFileSync(path.join(here, 'node_modules/axe-core/axe.min.js'), 'utf8');
fs.mkdirSync(out, { recursive: true });
const demos = (process.env.DEMOS || 'index.html,page.html,outside.html,say.html,shell.html').split(',');
const widths = [320, 390, 820, 1366, 1920];
const browser = await chromium.launch();
const results = [];
for (const file of demos) {
  const [base, hash] = file.split('#');
  const dir = (base === 'shell.html' || base === 'chat.html') ? here : path.join(root, 'demo');
  const url = ((base === 'shell.html' || base === 'chat.html') ? 'http://127.0.0.1:8812/' : 'http://127.0.0.1:8811/demo/') + base + (hash ? '#' + hash : '');
  for (const width of widths) for (const scheme of ['light', 'dark']) for (const coarse of [false, true]) {
    if (coarse && width > 820) continue;
    const ctx = await browser.newContext({ viewport: { width, height: 900 }, colorScheme: scheme, reducedMotion: 'reduce', hasTouch: coarse, isMobile: false });
    const page = await ctx.newPage();
    const errors = []; page.on('pageerror', e => errors.push(String(e))); page.on('console', m => m.type() === 'error' && m.text().includes('404') || errors.push(m.text()));
    await page.goto(url); await page.evaluate(() => document.fonts.ready);
    await page.addScriptTag({ content: axeSrc });
    const axe = await page.evaluate(async () => { const r = await axe.run(document, { runOnly: ['wcag2a', 'wcag2aa', 'wcag21a', 'wcag21aa', 'wcag22aa', 'best-practice'] }); return r.violations.map(v => ({ id: v.id, impact: v.impact, n: v.nodes.length, sample: v.nodes[0].target.join(' ') })); });
    const m = await page.evaluate(() => {
      const vis = el => { const r = el.getBoundingClientRect(); const cs = getComputedStyle(el); return r.width > 0 && r.height > 0 && cs.visibility !== 'hidden' && !el.closest('[hidden],dialog:not([open])'); };
      const sel = 'a[href],button,input:not([type=hidden]),select,textarea,summary,[role=tab],[role=button],[tabindex]:not([tabindex="-1"])';
      const label = el => (el.tagName.toLowerCase() + (el.className && typeof el.className === 'string' ? '.' + el.className.trim().split(/\s+/).slice(0, 2).join('.') : '') + ' "' + (el.getAttribute('aria-label') || el.textContent || '').trim().slice(0, 24) + '"');
      const small = [...document.querySelectorAll(sel)].filter(vis).map(el => { const r = el.getBoundingClientRect(); return { el: label(el), w: Math.round(r.width), h: Math.round(r.height), inline: getComputedStyle(el).display === 'inline' }; }).filter(x => !x.inline && (x.h < 44 || x.w < 44));
      const overflow = document.documentElement.scrollWidth - document.documentElement.clientWidth;
      const clipped = [...document.querySelectorAll('body *')].filter(vis).filter(el => { const r = el.getBoundingClientRect(); return r.right > innerWidth + 1 && !el.closest('.ui-strip,.ui-palette,[style*="overflow"]'); }).slice(0, 3).map(label);
      return { small, overflow, clipped };
    });
    // keyboard: Tab through, check the ring on each focused element
    await page.evaluate(() => document.activeElement?.blur()); const noRing = []; let count = 0; const seen = new Set();
    for (let i = 0; i < 150; i++) {
      await page.keyboard.press('Tab');
      const r = await page.evaluate(() => { const el = document.activeElement; if (!el || el === document.body) return null; const cs = getComputedStyle(el); const ring = cs.outlineStyle !== 'none' && parseFloat(cs.outlineWidth) > 0 || cs.boxShadow !== 'none'; const rc = el.getBoundingClientRect(); return { id: el.tagName.toLowerCase() + (el.className && typeof el.className === 'string' ? '.' + el.className.trim().split(/\s+/).slice(0, 2).join('.') : '') + ' "' + (el.getAttribute('aria-label') || el.textContent || '').trim().slice(0, 24) + '"', ring, key: [...document.querySelectorAll('*')].indexOf(el), visible: rc.width > 0 }; });
      if (!r) break; if (seen.has(r.key)) break; count++; if (!r.ring) noRing.push(r.id);
      if (seen.has(r.key)) break; seen.add(r.key);
    }
    const tag = `${file.replace('.html', '').replace('#','-')}-${width}-${scheme}${coarse ? '-touch' : ''}`;

// usage: node matrix.mjs <kit worktree>  - which states each component's CSS (src/*.css) carries, by selector
import fs from 'node:fs'; import path from 'node:path';
const root = process.argv[2];
const files = ['components.css', 'needs.css', 'shell.css', 'palette.css'];
const css = files.map(f => [f, fs.readFileSync(path.join(root, 'src', f), 'utf8')]);
